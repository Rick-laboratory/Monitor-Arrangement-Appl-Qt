cmake_minimum_required(VERSION 3.16)
project(MonitorArrangement VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
add_executable(monitor-arrangement WIN32
    src/main.cpp src/mainwindow.cpp src/mainwindow.h
    src/monitorarrangementwidget.cpp src/monitorarrangementwidget.h
    src/monitorinfo.h src/monitorlayout.h)
target_link_libraries(monitor-arrangement PRIVATE Qt${QT_VERSION_MAJOR}::Widgets)
if(MSVC)
    target_compile_options(monitor-arrangement PRIVATE /W4 /permissive-)
else()
    target_compile_options(monitor-arrangement PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(layout-tests tests/layout_tests.cpp)
    target_include_directories(layout-tests PRIVATE src)
    target_link_libraries(layout-tests PRIVATE Qt${QT_VERSION_MAJOR}::Core)
    add_test(NAME monitor-layout COMMAND layout-tests)
    add_test(NAME hidden-ui COMMAND monitor-arrangement --check)
    set_tests_properties(hidden-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
